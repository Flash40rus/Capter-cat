// ═══════════════════════════════════════════ // СИСТЕМА КОЛЛИЗИЙ (circle-based) // ═══════════════════════════════════════════ const HITBOX_FORGIVE = 0.88; // прощаем игроку 12% (мягче) let debugHitboxes = false; // Проверка пересечения двух кругов function circlesHit(a, b) { const dx = a.x - b.x; const dy = a.y - b.y; const r = a.r + b.r; return dx*dx + dy*dy < r*r; } // Хитбоксы кота — зависят от состояния function getCatHitboxes() { const cx = cat.x; const by = cat.y; // ─── ПОДКАТ: три круга вдоль тела, низко ─── if (cat.sliding) { return [ { x: cx - 22, y: by - 12, r: 13 }, { x: cx, y: by - 12, r: 13 }, { x: cx + 22, y: by - 12, r: 13 }, ]; } // ─── ОБЫЧНОЕ / ПРЫЖОК ─── const airborne = by < groundY - 2; const squash = airborne ? 0.94 : 1; // в прыжке чуть сжат по Y return [ // голова (сверху-справа) { x: cx + 28, y: by - 76 * squash, r: 14 }, // тело (центр) { x: cx - 3, y: by - 58 * squash, r: 20 }, // бёдра (сзади-снизу) { x: cx - 26, y: by - 48 * squash, r: 14 }, ]; } // Хитбоксы препятствия — по форме function getObstacleHitboxes(o) { const box = [ { x: o.x + o.w * 0.27, y: o.y + o.h / 2, r: Math.min(o.w, o.h) * 0.52 }, { x: o.x + o.w * 0.73, y: o.y + o.h / 2, r: Math.min(o.w, o.h) * 0.52 }, ]; switch (o.type) { case 'pepper': case 'cucumber': // Округлые → один круг return [{ x: o.x + o.w/2, y: o.y + o.h/2, r: Math.max(o.w, o.h) * 0.45 }]; case 'bone': // Вытянутая палочка → две «шишки» по краям return [ { x: o.x + o.w*0.15, y: o.y + o.h/2, r: o.h * 0.55 }, { x: o.x + o.w*0.85, y: o.y + o.h/2, r: o.h * 0.55 }, ]; case 'vacuum': // Широкий и низкий → два круга return [ { x: o.x + o.w*0.30, y: o.y + o.h/2, r: o.h * 0.55 }, { x: o.x + o.w*0.70, y: o.y + o.h/2, r: o.h * 0.55 }, ]; case 'slipper': case 'plant': // Компактные → один круг return [{ x: o.x + o.w/2, y: o.y + o.h/2, r: Math.min(o.w, o.h) * 0.52 }]; case 'book': // Тонкая и широкая → два сплюснутых круга return [ { x: o.x + o.w*0.30, y: o.y + o.h/2, r: o.h * 0.62 }, { x: o.x + o.w*0.70, y: o.y + o.h/2, r: o.h * 0.62 }, ]; case 'box': default: // Коробка → две «колонны» return box; } } // Главная проверка function checkCollisions() { if (G.invuln > 0) return; // неуязвим после удара const catBoxes = getCatHitboxes(); // прощение — уменьшаем радиус кругов кота for (const cb of catBoxes) cb.r *= HITBOX_FORGIVE; for (let i = 0; i < G.obstacles.length; i++) { const o = G.obstacles[i]; const obs = getObstacleHitboxes(o); for (const ob of obs) { for (const cb of catBoxes) { if (circlesHit(cb, ob)) { hit(); o.hitFlash = 8; // вспышка на препятствии return; } } } } } // Сбор рыбок — тоже circle-based, но с щедрым радиусом function checkItemPickup() { const cx = cat.x; const by = cat.y; const catCenter = cat.sliding ? { x: cx, y: by - 12 } : { x: cx, y: by - 58 }; const pickupR = cat.sliding ? 34 : 44; for (const it of G.items) { if (it.collected) continue; const dx = catCenter.x - it.x; const dy = catCenter.y - it.y; if (dx*dx + dy*dy < (pickupR + it.r) * (pickupR + it.r)) { it.collected = true; G.distance += 120; burst(it.x, it.y, '#f4b942', 8, 2.5); } } G.items = G.items.filter(i => !i.collected); } // Debug: рисование хитбоксов window.addEventListener('keydown', (e) => { if (e.code === 'KeyD') debugHitboxes = !debugHitboxes; }); function drawHitboxes() { if (!debugHitboxes) return; ctx.save(); // хитбоксы кота if (G.started && !G.dead) { const cbs = getCatHitboxes(); for (const cb of cbs) cb.r *= HITBOX_FORGIVE; ctx.strokeStyle = '#00ff88'; ctx.lineWidth = 2; for (const cb of cbs) { ctx.beginPath(); ctx.arc(cb.x, cb.y, cb.r, 0, Math.PI*2); ctx.stroke(); } } // хитбоксы препятствий ctx.strokeStyle = '#ff3355'; for (const o of G.obstacles) { const obs = getObstacleHitboxes(o); for (const ob of obs) { ctx.beginPath(); ctx.arc(ob.x, ob.y, ob.r, 0, Math.PI*2); ctx.stroke(); } } // рыбки ctx.strokeStyle = '#ffdd55'; for (const it of G.items) { ctx.beginPath(); ctx.arc(it.x, it.y, it.r, 0, Math.PI*2); ctx.stroke(); } ctx.restore(); }
